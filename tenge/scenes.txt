function banknote(x, y, w, h, rot, kind, a = 1, sc = 1) {   // flat note: 'rub' soviet ruble, 'tng' tenge
  if (a <= 0 || sc <= 0) return; ctx.save(); ctx.globalAlpha = a; ctx.translate(x, y); ctx.rotate(rot); ctx.scale(sc, sc);
  const R = kind === 'rub'; const bg = R ? '#c9b98a' : F.am, ink = R ? '#7a5c3a' : F.dk, ac = R ? '#a8875a' : F.tq;
  ctx.fillStyle = F.sh; rr(-w / 2 + 5, -h / 2 + 6, w, h, h * .08); ctx.fill();
  ctx.fillStyle = bg; rr(-w / 2, -h / 2, w, h, h * .08); ctx.fill();
  ctx.strokeStyle = ink; ctx.lineWidth = Math.max(2, h * .04); rr(-w / 2 + h * .1, -h / 2 + h * .1, w - h * .2, h * .8, h * .05); ctx.stroke();
  ctx.fillStyle = ac; ctx.beginPath(); ctx.arc(w * .22, 0, h * .28, 0, 7); ctx.fill();
  if (R) { ctx.fillStyle = ink; star(w * .22, 0, h * .18, 1, ink); txt('РУБЛЬ', -w * .12, 0, h * .24, ink, 'MB', { blur: 0, dy: 0 }); }
  else { txt('₸', w * .22, h * .02, h * .4, F.dk, 'MB', { blur: 0, dy: 0 }); ctx.strokeStyle = F.dk; ctx.lineWidth = Math.max(2, h * .035);
    for (const sx of [-1, 1]) { ctx.beginPath(); ctx.moveTo(-w * .18, 0); ctx.bezierCurveTo(-w * .18 + sx * h * .2, -h * .25, -w * .18 + sx * h * .35, -h * .05, -w * .18 + sx * h * .25, h * .1); ctx.stroke(); }
    txt('ТЕҢГЕ', -w * .18, h * .27, h * .14, F.dk, 'MB', { blur: 0, dy: 0 }); }
  ctx.restore();
}
function stampMark(s, x, y, p, size = 70, rot = -.12, col = F.co) {   // rubber stamp slams in
  if (p <= 0) return; const k = 1 + 1.4 * (1 - eout(clamp(p))); const w = tw(s, size, 'MB') + size * .9, h = size * 1.6;
  ctx.save(); ctx.globalAlpha = clamp(p * 3); ctx.translate(x, y); ctx.rotate(rot); ctx.scale(k, k);
  ctx.strokeStyle = col; ctx.lineWidth = size * .1; rr(-w / 2, -h / 2, w, h, size * .18); ctx.stroke(); ctx.lineWidth = size * .04; rr(-w / 2 + size * .16, -h / 2 + size * .16, w - size * .32, h - size * .32, size * .1); ctx.stroke();
  ctx.fillStyle = 'rgba(239,71,111,.12)'; rr(-w / 2, -h / 2, w, h, size * .18); ctx.fill();
  txt(s, 0, size * .05, size, col, 'MB', { blur: 0, dy: 0 }); ctx.restore();
}
function suitcase(x, y, s, open, glare = 1, glow = 0) {   // (x,y) = bottom centre; open 0..1 lifts the lid
  ctx.save(); ctx.translate(x, y); ctx.scale(s, s);
  const W2 = 300, Hb = 250, oc = Math.cos(clamp(open) * Math.PI * .55);
  // body
  shade(c => { ctx.fillStyle = c || '#2a3a6e'; rr(-W2, -Hb, W2 * 2, Hb, 26); ctx.fill(); if (!c) { ctx.fillStyle = 'rgba(0,0,0,.18)'; ctx.fillRect(-W2, -Hb * .52, W2 * 2, 14);
    ctx.fillStyle = F.am; for (const sx of [-1, 1]) { rr(sx * 200 - 28, -Hb - 6, 56, 46, 8); ctx.fill(); } } }, 12, 14);
  // money inside (visible when open)
  if (open > .05) {
    ctx.save(); ctx.beginPath(); ctx.rect(-W2 + 20, -Hb - 10, W2 * 2 - 40, 60); ctx.clip();
    for (let r = 0; r < 2; r++) for (let i = 0; i < 4; i++) banknote(-210 + i * 140, -Hb + 26 - r * 18, 130, 64, (i % 2 ? .04 : -.04), 'tng', 1);
    ctx.restore();
    const gl = glare * clamp(open * 2); if (gl > 0) { ctx.save(); ctx.globalAlpha = gl; const g = ctx.createRadialGradient(0, -Hb, 10, 0, -Hb, 380); g.addColorStop(0, 'rgba(255,255,240,1)'); g.addColorStop(.35, 'rgba(255,240,200,.85)'); g.addColorStop(1, 'rgba(255,230,160,0)');
      ctx.fillStyle = g; ctx.beginPath(); ctx.ellipse(0, -Hb, 420, 200, 0, 0, 7); ctx.fill(); ctx.restore(); }
    if (glow > 0) { ctx.save(); ctx.globalAlpha = glow; ctx.fillStyle = GOLD; ctx.shadowColor = GOLD; ctx.shadowBlur = 80; ctx.beginPath(); ctx.moveTo(-W2 + 30, -Hb); ctx.lineTo(-W2 - 200, -Hb - 900); ctx.lineTo(W2 + 200, -Hb - 900); ctx.lineTo(W2 - 30, -Hb); ctx.closePath(); ctx.globalAlpha = glow * .25; ctx.fill(); ctx.restore(); }
  }
  // lid: hinged at the back edge (y=-Hb), drawn foreshortened
  const lh = Hb * .9 * oc; ctx.save(); ctx.translate(0, -Hb);
  if (open < .5) { shade(c => { ctx.fillStyle = c || '#334680'; rr(-W2, -lh * (open > 0 ? 1 : 0) - 0, W2 * 2, Math.max(8, lh * (1 - clamp(open * 2)) + 8), 22); ctx.fill(); }, 0, 0); }
  else { ctx.fillStyle = '#1e2a55'; ctx.beginPath(); ctx.moveTo(-W2, 0); ctx.lineTo(-W2 + 30, -Hb * .8 * clamp((open - .5) * 2)); ctx.lineTo(W2 - 30, -Hb * .8 * clamp((open - .5) * 2)); ctx.lineTo(W2, 0); ctx.closePath(); ctx.fill(); }
  ctx.restore();
  // handle
  ctx.strokeStyle = F.dk; ctx.lineWidth = 18; ctx.lineCap = 'round'; ctx.beginPath(); ctx.moveTo(-70, -Hb * (open > .05 ? 1 : 1) + 4); ctx.lineTo(-70, -Hb + 4); ctx.stroke();
  ctx.restore();
}
